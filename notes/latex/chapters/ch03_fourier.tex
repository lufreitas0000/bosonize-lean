\chapter{The Finite Fourier Transform}
\label{ch:fourier}

\section{Introduction and Algebraic Setting}

The discrete Fourier transform is the central kinematic bridge of quantum field theory on a lattice. It connects the spatial coordinate representation on the periodic lattice $\Lambda = \mathbb{Z}/L\mathbb{Z}$ to the dual momentum representation on the centered band $\Lambda^* = \{-h+1, \dots, h\}$ (for even $L = 2h$).

In standard physics presentations, the Fourier transform is defined immediately on complex functions using transcendental exponentials $e^{2\pi i k x / L}$ and the unitary normalization factor $1/\sqrt{L}$. While physically standard, this conflates two distinct mathematical layers:
\begin{enumerate}
    \item \textbf{The algebraic character layer:} Character orthogonality and algebraic Fourier inversion require only a commutative field $K$ containing a primitive $L$-th root of unity $\zeta$, and the invertibility of the lattice size ($L \cdot 1_K \neq 0$, meaning the characteristic of $K$ does not divide $L$). Square roots, analytic limits, and metric structures are completely unnecessary at this level.
    \item \textbf{The Hilbert unitary layer:} To make the Fourier transform an exact isometry that preserves Canonical Anticommutation Relations (CAR) between position and momentum fermions without spurious factors of $\sqrt{L}$, we instantiate $K = \mathbb{C}$ and introduce the metric normalization $a = 1/\sqrt{L}$.
\end{enumerate}

In this chapter, we develop both layers in full mathematical rigor, following the verified constructive formalization.

\section{Generic Character Algebra over Fields}

\begin{definition}[Primitive Root of Unity]
Let $K$ be a commutative field, and let $L \ge 1$ be a positive integer. An element $\zeta \in K$ is a primitive $L$-th root of unity if:
\begin{equation}
\zeta^L = 1, \qquad \text{and} \qquad \zeta^m \neq 1 \quad \text{for all } m \in \{1, 2, \dots, L-1\}.
\end{equation}
\end{definition}

When $K = \mathbb{C}$, the canonical primitive $L$-th root of unity is:
\begin{equation}
\zeta_{\mathrm{can}} \coloneqq \exp\left(\frac{2\pi i}{L}\right).
\end{equation}

\begin{definition}[Representative-Independent Character]
For any momentum mode $k \in \Lambda^*$ and spatial site $x \in \Lambda$, let $x_{\mathrm{val}} \in \{0, 1, \dots, L-1\}$ denote the unique unsigned residue of $x$. The discrete Fourier character is defined by:
\begin{equation}
\chi(k, x) \coloneqq \zeta^{k \cdot x_{\mathrm{val}}}.
\end{equation}
\end{definition}

\begin{lemma}[Representative Independence]
\label{lem:rep_independence}
For any integer $k \in \mathbb{Z}$ and any integer representative $\tilde{x} \in \mathbb{Z}$ of $x$ (so that $\tilde{x} \equiv x_{\mathrm{val}} \pmod L$):
\begin{equation}
\zeta^{k \tilde{x}} = \zeta^{k \cdot x_{\mathrm{val}}} = \chi(k, x).
\end{equation}
In particular, $\chi$ depends only on the quotient residue class $x \in \mathbb{Z}/L\mathbb{Z}$.
\end{lemma}

\begin{proof}
Since $\tilde{x} \equiv x_{\mathrm{val}} \pmod L$, there exists $m \in \mathbb{Z}$ such that $\tilde{x} = x_{\mathrm{val}} + m L$. Therefore:
\[
\zeta^{k \tilde{x}} = \zeta^{k (x_{\mathrm{val}} + m L)} = \zeta^{k x_{\mathrm{val}}} \cdot (\zeta^L)^{k m} = \zeta^{k x_{\mathrm{val}}} \cdot 1^{k m} = \zeta^{k x_{\mathrm{val}}}.
\]
\end{proof}

\section{The Unscaled Analysis and Synthesis Transforms}

Because the spatial lattice $\Lambda$ and the momentum band $\Lambda^*$ are distinct sets, the Fourier transform consists of two directional linear maps between distinct function modules:
\begin{itemize}
    \item Spatial function space: $K^\Lambda \equiv (\Lambda \to K)$.
    \item Momentum function space: $K^{\Lambda^*} \equiv (\Lambda^* \to K)$.
\end{itemize}

\begin{definition}[Analysis and Synthesis Transforms]
We define the $K$-linear transforms:
\begin{enumerate}
    \item \textbf{Analysis Transform $S : K^\Lambda \to K^{\Lambda^*}$:}
    \begin{equation}
    (S f)(k) \coloneqq \sum_{x \in \Lambda} f(x)\, \chi(-k, x), \quad k \in \Lambda^*.
    \end{equation}
    \item \textbf{Synthesis Transform $T : K^{\Lambda^*} \to K^\Lambda$:}
    \begin{equation}
    (T g)(x) \coloneqq \sum_{k \in \Lambda^*} g(k)\, \chi(k, x), \quad x \in \Lambda.
    \end{equation}
\end{enumerate}
\end{definition}

Note that $\chi(-k, x) = \zeta^{-k x_{\mathrm{val}}} = (\zeta^{k x_{\mathrm{val}}})^{-1} = \chi(k, x)^{-1}$. By Lemma~\ref{lem:rep_independence}, evaluating at $-k \in \mathbb{Z}$ agrees with evaluating at the transported band negation $\ominus k \in \Lambda^*$, ensuring algebraic consistency even when $-k \notin \Lambda^*$.

\section{Dual Character Orthogonality and Algebraic Inversion}

The foundation of Fourier analysis on finite abelian groups is character orthogonality.

\begin{lemma}[Geometric Sum of Roots of Unity]
\label{lem:geom_sum}
Let $\zeta \in K$ be a primitive $L$-th root of unity. For any integer $m \in \mathbb{Z}$:
\begin{equation}
\sum_{x=0}^{L-1} \zeta^{m x} = \begin{cases}
L \cdot 1_K, & \text{if } m \equiv 0 \pmod L, \\
0, & \text{if } m \not\equiv 0 \pmod L.
\end{cases}
\end{equation}
\end{lemma}

\begin{proof}
If $m \equiv 0 \pmod L$, then $\zeta^m = 1$, giving $\sum_{x=0}^{L-1} 1 = L \cdot 1_K$.
If $m \not\equiv 0 \pmod L$, then $\zeta^m \neq 1$. Telescoping yields:
\[
(1 - \zeta^m) \sum_{x=0}^{L-1} \zeta^{m x} = 1 - \zeta^{m L} = 1 - 1 = 0.
\]
Because $K$ is a field and $(1 - \zeta^m) \neq 0$, dividing by $(1 - \zeta^m)$ proves the sum is $0$.
\end{proof}

\begin{theorem}[Dual Character Orthogonality]
\label{thm:fourier_orthogonality}
Let $K$ be a field with a primitive $L$-th root of unity $\zeta$.
\begin{enumerate}
    \item \textbf{Spatial Orthogonality:} For any momenta $k, p \in \Lambda^*$:
    \begin{equation}
    \sum_{x \in \Lambda} \chi(k, x)\, \chi(-p, x) = (L \cdot 1_K)\, \delta_{k, p}.
    \end{equation}
    \item \textbf{Momentum Orthogonality:} For any spatial sites $x, y \in \Lambda$:
    \begin{equation}
    \sum_{k \in \Lambda^*} \chi(k, x)\, \chi(-k, y) = (L \cdot 1_K)\, \delta_{x, y}.
    \end{equation}
\end{enumerate}
\end{theorem}

\begin{proof}
For spatial orthogonality, $\chi(k, x) \chi(-p, x) = \zeta^{(k-p) x_{\mathrm{val}}}$. By Lemma~\ref{lem:geom_sum}, the sum equals $L \cdot 1_K$ if $k - p \equiv 0 \pmod L$, and $0$ otherwise. Because $k, p \in \Lambda^*$, their difference satisfies $-L < k - p < L$. The only multiple of $L$ in this range is $0$, which implies $k = p$. Thus the sum is $(L \cdot 1_K) \delta_{k, p}$.
Momentum orthogonality follows similarly: $\sum_{k \in \Lambda^*} \zeta^{k (x_{\mathrm{val}} - y_{\mathrm{val}})}$. Translating the sum over the centered band $\Lambda^* = \{-h+1, \dots, h\}$ to $\{0, \dots, L-1\}$ introduces an overall phase factor $\zeta^{(-h+1)(x_{\mathrm{val}} - y_{\mathrm{val}})}$ that is non-zero in $K$, leaving the sum proportional to $(L \cdot 1_K) \delta_{x, y}$.
\end{proof}

\begin{theorem}[Two-Sided Algebraic Inversion]
\label{thm:algebraic_inversion}
Assume $L \cdot 1_K \neq 0$ in $K$. Then the transforms $S$ and $T$ satisfy:
\begin{align}
T \circ S &= (L \cdot 1_K)\, I_{K^\Lambda}, \\
S \circ T &= (L \cdot 1_K)\, I_{K^{\Lambda^*}}.
\end{align}
Consequently, both $S$ and $T$ are vector space isomorphisms.
\end{theorem}

\begin{proof}
Let $f \in K^\Lambda$ and evaluate $(T(S f))(x)$:
\begin{align*}
(T(S f))(x) &= \sum_{k \in \Lambda^*} (S f)(k)\, \chi(k, x) = \sum_{k \in \Lambda^*} \left( \sum_{y \in \Lambda} f(y)\, \chi(-k, y) \right) \chi(k, x) \\
&= \sum_{y \in \Lambda} f(y) \left( \sum_{k \in \Lambda^*} \chi(k, x)\, \chi(-k, y) \right).
\end{align*}
By Theorem~\ref{thm:fourier_orthogonality}(2), the inner sum is $(L \cdot 1_K) \delta_{x, y}$. Thus $(T(S f))(x) = (L \cdot 1_K) f(x)$.
Similarly, $(S(T g))(k) = (L \cdot 1_K) g(k)$ follows from spatial orthogonality.
\end{proof}

\section{Unitary Fourier Transform in the Hilbert Setting}

When $K = \mathbb{C}$, the function spaces $K^\Lambda$ and $K^{\Lambda^*}$ are complex Euclidean Hilbert spaces equipped with standard inner products:
\begin{equation}
\langle f_1, f_2 \rangle_{\mathrm{pos}} \coloneqq \sum_{x \in \Lambda} \overline{f_1(x)}\, f_2(x), \qquad \langle g_1, g_2 \rangle_{\mathrm{mom}} \coloneqq \sum_{k \in \Lambda^*} \overline{g_1(k)}\, g_2(k).
\end{equation}

\begin{definition}[Unitary Fourier Transform]
Let $a \coloneqq 1/\sqrt{L} \in \mathbb{R}_{>0}$. The \emph{unitary discrete Fourier transform} $U : \mathbb{C}^\Lambda \to \mathbb{C}^{\Lambda^*}$ is defined by:
\begin{equation}
U \coloneqq \frac{1}{\sqrt{L}} S, \qquad (U f)(k) = \frac{1}{\sqrt{L}} \sum_{x \in \Lambda} f(x)\, \chi(-k, x).
\end{equation}
Its inverse is $U^{-1} = \frac{1}{\sqrt{L}} T$.
\end{definition}

\begin{theorem}[Unitary Isometry and Plancherel Identity]
\label{thm:unitary_plancherel}
\begin{enumerate}
    \item $U^\dagger = U^{-1}$. That is, $U$ is a unitary operator.
    \item \textbf{Parseval/Plancherel Identity:} For all functions $f_1, f_2 \in \mathbb{C}^\Lambda$:
    \begin{equation}
    \langle U f_1, U f_2 \rangle_{\mathrm{mom}} = \langle f_1, f_2 \rangle_{\mathrm{pos}}.
    \end{equation}
\end{enumerate}
\end{theorem}

\begin{proof}
We evaluate the adjoint directly:
\begin{align*}
\langle S f, g \rangle_{\mathrm{mom}} &= \sum_{k \in \Lambda^*} \overline{(S f)(k)}\, g(k) = \sum_{k \in \Lambda^*} \left( \sum_{x \in \Lambda} \overline{f(x)}\, \overline{\chi(-k, x)} \right) g(k) \\
&= \sum_{x \in \Lambda} \overline{f(x)} \left( \sum_{k \in \Lambda^*} g(k)\, \chi(k, x) \right) = \sum_{x \in \Lambda} \overline{f(x)}\, (T g)(x) = \langle f, T g \rangle_{\mathrm{pos}},
\end{align*}
using $\overline{\chi(-k, x)} = \chi(k, x)$. Thus $S^\dagger = T$.
Scaling by $a = 1/\sqrt{L} \in \mathbb{R}$:
\[
U^\dagger = (a S)^\dagger = a S^\dagger = a T = U^{-1}.
\]
Then $\langle U f_1, U f_2 \rangle_{\mathrm{mom}} = \langle f_1, U^\dagger U f_2 \rangle_{\mathrm{pos}} = \langle f_1, f_2 \rangle_{\mathrm{pos}}$, proving that $U$ is a unitary isometry.
\end{proof}

\section{Diagonalization of Difference Operators and Exact Dispersion}

We now connect the Fourier transform to the discrete difference operators of Chapter~\ref{ch:umbral_calculus}.

\begin{theorem}[Diagonal Action on Characters]
\label{thm:diagonal_dispersion}
Let $\chi_k(x) \coloneqq \chi(k, x)$ denote the Fourier character as a spatial function.
\begin{enumerate}
    \item The shift operator $E$ acts diagonally:
    \begin{equation}
    (E \chi_k)(x) = \zeta^k \chi_k(x).
    \end{equation}
    \item The forward and backward differences act diagonally:
    \begin{align}
    (\Delta \chi_k)(x) &= (\zeta^k - 1) \chi_k(x), \\
    (\nabla \chi_k)(x) &= (1 - \zeta^{-k}) \chi_k(x).
    \end{align}
    \item The discrete Laplacian acts diagonally with eigenvalue $\lambda_k = \zeta^k + \zeta^{-k} - 2$:
    \begin{equation}
    (\Delta\nabla \chi_k)(x) = (\zeta^k + \zeta^{-k} - 2) \chi_k(x).
    \end{equation}
\end{enumerate}
\end{theorem}

\begin{proof}
Evaluating the definitions:
\begin{enumerate}
    \item $(E \chi_k)(x) = \chi_k(x+1) = \zeta^{k (x_{\mathrm{val}}+1)} = \zeta^k \zeta^{k x_{\mathrm{val}}} = \zeta^k \chi_k(x)$ by Lemma~\ref{lem:rep_independence}.
    \item $(\Delta \chi_k)(x) = (E - I)\chi_k(x) = (\zeta^k - 1)\chi_k(x)$, and $(\nabla \chi_k)(x) = (I - E^{-1})\chi_k(x) = (1 - \zeta^{-k})\chi_k(x)$.
    \item Composing the two differences:
    \[
    \Delta(\nabla \chi_k) = (\zeta^k - 1)(1 - \zeta^{-k})\chi_k = (\zeta^k - 1 - 1 + \zeta^{-k})\chi_k = (\zeta^k + \zeta^{-k} - 2)\chi_k.
    \]
\end{enumerate}
\end{proof}

\begin{corollary}[Exact Lattice Dispersion in $\mathbb{C}$]
Under the canonical root $\zeta = e^{2\pi i / L}$, the Laplacian eigenvalue evaluates to:
\begin{equation}
\zeta^k + \zeta^{-k} - 2 = -4 \sin^2\left(\frac{\pi k}{L}\right).
\end{equation}
\end{corollary}

\begin{proof}
Using Euler's formula with angle $\theta = \frac{2\pi k}{L}$:
\[
\zeta^k + \zeta^{-k} = e^{i\theta} + e^{-i\theta} = 2 \cos\theta.
\]
Using the trigonometric identity $\cos\theta = 1 - 2 \sin^2(\theta/2)$:
\[
\zeta^k + \zeta^{-k} - 2 = 2 \cos\left(\frac{2\pi k}{L}\right) - 2 = 2 \left( 1 - 2 \sin^2\left(\frac{\pi k}{L}\right) \right) - 2 = -4 \sin^2\left(\frac{\pi k}{L}\right).
\]
\end{proof}

\paragraph{Formal Correctness vs.\ Physical Implications: Trigonometric Truncation vs.\ Continuum Dispersion.}
In continuum physics, the spatial kinetic operator is $-\partial_x^2$, whose Fourier transform yields the quadratic dispersion $p_{\mathrm{phys}}^2 = (2\pi k / L)^2$. In the long-wavelength continuum limit where the mode index is small compared to the lattice size ($|k| \ll L$), we approximate $\sin(\pi k / L) \approx \frac{\pi k}{L}$, and the exact lattice dispersion becomes:
\begin{equation}
-4 \sin^2\left(\frac{\pi k}{L}\right) \approx -4 \left(\frac{\pi k}{L}\right)^2 = -\left(\frac{2\pi k}{L}\right)^2 = -p_{\mathrm{phys}}^2,
\end{equation}
recovering the continuous kinetic Laplacian $-\partial_x^2$. 

On the discrete lattice, the trigonometric dispersion $-4 \sin^2(\pi k / L)$ is bounded between $-4$ and $0$ for all modes in the Brillouin zone. High-momentum modes near the Brillouin zone edge do not diverge quadratically; their energy saturates at $-4$. This natural trigonometric bending provides an intrinsic, exact ultraviolet regularization that eliminates divergences without needing arbitrary momentum cutoffs.

\section{Chapter Summary and Literature Comparison}

In this chapter, we developed the finite Fourier transform bridging spatial and momentum representations:
\begin{itemize}
    \item The algebraic character layer is defined over any commutative field $K$ containing a primitive $L$-th root $\zeta$, requiring only $L \cdot 1_K \neq 0$.
    \item Dual character orthogonality yields two-sided algebraic inversion $T \circ S = S \circ T = L I$.
    \item In the complex Hilbert setting ($K = \mathbb{C}$), scaling by $1/\sqrt{L}$ yields the unitary Fourier isometry $U$, satisfying the Parseval/Plancherel identity.
    \item The discrete difference operators and Laplacian $\Delta\nabla$ are diagonalized by characters, yielding the exact bounded lattice dispersion $-4\sin^2(\pi k / L)$.
\end{itemize}

\paragraph{Comparison with Standard Continuum Literature.}
In standard condensed matter treatments of 1D systems, such as Giamarchi~\cite{giamarchi2004}, Gogolin et al.~\cite{gogolin1998}, and von Delft and Schoeller~\cite{vondelft1998}, the Fourier transform is formulated with continuous plane waves $e^{i k x}$, integrating over a continuous wire $[0, \mathcal{L}]$. Field commutators are delta functions $[\psi(x), \psi^\dagger(y)] = \delta(x-y)$, which are singular distributions requiring smearing with test functions.

In our constructive framework, all transforms are finite sums over the discrete group $\Lambda$. Kronecker deltas $\delta_{x, y}$ replace Dirac delta distributions, characters $\chi(k, x)$ are finite algebraic roots of unity, and the Fourier transform is a strictly unitary operator on finite-dimensional spaces. This ensures that when we construct spatial fermionic operators in Chapter~\ref{ch:fermions}, the CAR relations are preserved exactly with zero distribution-theoretic artifacts.
