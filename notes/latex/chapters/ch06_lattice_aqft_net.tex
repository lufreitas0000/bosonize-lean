\chapter{The Lattice AQFT Net and Boundary Extensions}
\label{ch:lattice_aqft_net}

\section{Introduction to Lattice AQFT and Algebraic Preliminaries}

In standard formulations of Quantum Field Theory, fields $\psi(x)$ are regarded as primary objects, and physical states are defined as rays in a Hilbert space. In Algebraic Quantum Field Theory (AQFT), following the foundational framework of Rudolf Haag and Daniel Kastler~\cite{haag1996}, the primary mathematical structure of a quantum theory is its \emph{net of local observable algebras}: an assignment of a unital $*$-algebra $\mathfrak{A}(O)$ to each spacetime region $O$, satisfying causality, isotony, and covariance.

In continuum spacetime, establishing the existence of a local net that satisfies the Haag--Kastler axioms for interacting theories is an arduous analytic problem, often encountering severe ultraviolet singularities. On the discrete spatial lattice $\Lambda = \mathbb{Z}/L\mathbb{Z}$ ($L = 2h$, $h \ge 1$), the situation is dramatically different and conceptually pristine:
\begin{enumerate}
    \item No abstract physical axioms need to be postulated.
    \item Every local algebra $\mathfrak{A}(I)$ is constructed explicitly from the spatial CAR operators $\{\psi_x, \psi_x^\dagger \mid x \in I\}$ acting on the finite-dimensional fermionic Fock space $\mathcal{F}(L)$.
    \item All the Haag--Kastler axioms---including isotony, additivity, twisted locality, and global irreducibility---are established as rigorous constructive theorems.
\end{enumerate}

\begin{remark}[Algebraic Primer for Physicists]
To ensure the mathematical terminology is completely transparent to readers from both mathematics and theoretical physics, we review the essential algebraic concepts used in this chapter:
\begin{itemize}
    \item \textbf{Endomorphism Algebra $\mathrm{End}_{\mathbb{C}}(V)$:} For a complex vector space $V$, $\mathrm{End}_{\mathbb{C}}(V)$ is the associative algebra of all linear maps from $V$ to $V$. When $V = \mathcal{F}(L)$ has finite dimension $2^L$, $\mathrm{End}_{\mathbb{C}}(V)$ is simply the algebra of $2^L \times 2^L$ complex matrices $\mathrm{Mat}_{2^L \times 2^L}(\mathbb{C})$.
    \item \textbf{$C^*$-Algebra and $*$-Subalgebra:} A complex associative algebra $\mathcal{A}$ endowed with an involutive conjugate-linear anti-automorphism $* : \mathcal{A} \to \mathcal{A}$ satisfying $(A^*) = A$, $(A B)^* = B^* A^*$, and the $C^*$-norm property $\|A^* A\| = \|A\|^2$. On a finite-dimensional Hilbert space, the $*$-operation is the standard Hermitian adjoint $A^* \equiv A^\dagger$, and every $*$-subalgebra is automatically closed and forms a $C^*$-algebra.
    \item \textbf{Unital Subalgebra:} A subalgebra $\mathcal{B} \subseteq \mathcal{A}$ is \emph{unital} if it contains the multiplicative identity operator $I$ of the ambient algebra $\mathcal{A}$.
    \item \textbf{Generated Subalgebra $\mathrm{AlgGen}_{\mathbb{C}}(\mathcal{S})$:} For any subset of operators $\mathcal{S} \subseteq \mathrm{End}_{\mathbb{C}}(V)$, $\mathrm{AlgGen}_{\mathbb{C}}(\mathcal{S})$ denotes the smallest unital complex subalgebra containing $\mathcal{S}$. It consists of all finite complex linear combinations of finite products (words) of operators in $\mathcal{S}$ and the identity $I$.
    \item \textbf{Automorphism:} An invertible linear map $\alpha : \mathcal{A} \to \mathcal{A}$ preserving algebraic multiplication: $\alpha(A B) = \alpha(A) \alpha(B)$. It is a $*$-automorphism if it also preserves the adjoint: $\alpha(A^\dagger) = (\alpha(A))^\dagger$.
\end{itemize}
\end{remark}

\section{The Local Net of Observable Algebras}

Let $\Lambda = \mathbb{Z}/L\mathbb{Z}$ with $L = 2h$ be the discrete spatial lattice, and let $\mathcal{F}(L)$ be the fermionic Fock space of dimension $2^L$. The ambient operator algebra is the finite-dimensional $C^*$-algebra $\mathrm{End}_{\mathbb{C}}(\mathcal{F}(L))$; all fermionic fields and macroscopic operators introduced below act on $\mathcal{F}(L)$ unless a different domain is stated.

\begin{definition}[Local Generating Sets and Local Algebras]
For any spatial subset (region) $I \subseteq \Lambda$:
\begin{enumerate}
    \item The set of \emph{local fermionic generators} $\mathcal{G}(I)$ is defined by:
    \begin{equation}
    \mathcal{G}(I) \coloneqq \{ \psi_x \mid x \in I \} \cup \{ \psi_x^\dagger \mid x \in I \}.
    \end{equation}
    \item The \emph{local field algebra} $\mathfrak{A}(I)$ is the unital complex subalgebra of $\mathrm{End}_{\mathbb{C}}(\mathcal{F}(L))$ generated by $\mathcal{G}(I)$:
    \begin{equation}
    \mathfrak{A}(I) \coloneqq \mathrm{AlgGen}_{\mathbb{C}}(\mathcal{G}(I)).
    \end{equation}
    Because $(\psi_x)^\dagger = \psi_x^\dagger$ and $(\psi_x^\dagger)^\dagger = \psi_x$, $\mathcal{G}(I)$ is closed under the Hilbert space adjoint. Consequently, $\mathfrak{A}(I)$ is automatically a unital $*$-subalgebra of $\mathrm{End}_{\mathbb{C}}(\mathcal{F}(L))$.
    \item Over the empty set, the algebra consists solely of scalar multiples of the identity:
    \begin{equation}
    \mathfrak{A}(\emptyset) = \mathbb{C} \cdot I.
    \end{equation}
\end{enumerate}
\end{definition}

\section[Global Parity and the Z2-Grading]{Global Parity and the $\mathbb{Z}_2$-Grading}

Fermionic operators change sign under a $2\pi$ spatial rotation, corresponding to the global fermion parity symmetry.

\begin{definition}[Parity Automorphism]
Let $\Gamma \in \mathrm{End}_{\mathbb{C}}(\mathcal{F}(L))$ be the global fermion parity operator from Chapter~\ref{ch:car_fock}, satisfying $\Gamma^\dagger = \Gamma$, $\Gamma^2 = I$, and $\Gamma \psi_x \Gamma = -\psi_x$, $\Gamma \psi_x^\dagger \Gamma = -\psi_x^\dagger$ for all $x \in \Lambda$. The \emph{parity automorphism} $\alpha : \mathrm{End}_{\mathbb{C}}(\mathcal{F}(L)) \to \mathrm{End}_{\mathbb{C}}(\mathcal{F}(L))$ is defined by:
\begin{equation}
\alpha(A) \coloneqq \Gamma A \Gamma.
\end{equation}
Because $\Gamma^2 = I$, $\alpha$ is an involutive $*$-automorphism: $\alpha(\alpha(A)) = A$ and $\alpha(A B) = \alpha(A) \alpha(B)$.
\end{definition}

\begin{definition}[Homogeneous Subspaces and Observable Net]
For each parity degree $\sigma \in \{0, 1\} \equiv \mathbb{Z}_2$, the $\sigma$-homogeneous local subspace $\mathfrak{A}_\sigma(I)$ is defined by:
\begin{equation}
\mathfrak{A}_\sigma(I) \coloneqq \{ A \in \mathfrak{A}(I) \mid \alpha(A) = (-1)^\sigma A \}.
\end{equation}
\begin{enumerate}
    \item \textbf{Even Subalgebra (Physical Observables):} $\mathfrak{A}_+(I) \coloneqq \mathfrak{A}_0(I) = \{A \in \mathfrak{A}(I) \mid \alpha(A) = A\}$.
    Because $\alpha(A B) = \alpha(A) \alpha(B) = A B$, $\mathfrak{A}_+(I)$ is a unital $*$-subalgebra. In quantum physics, only even operators are gauge-invariant under fermion parity and correspond to physically measurable observables (such as charge densities $\psi_x^\dagger \psi_x$ and hopping energies $\psi_x^\dagger \psi_{x+1} + \psi_{x+1}^\dagger \psi_x$).
    \item \textbf{Odd Subspace (Fermionic Fields):} $\mathfrak{A}_-(I) \coloneqq \mathfrak{A}_1(I) = \{A \in \mathfrak{A}(I) \mid \alpha(A) = -A\}$.
    This is a linear subspace, but not an algebra, since the product of two odd operators is even: $\mathfrak{A}_-(I) \cdot \mathfrak{A}_-(I) \subseteq \mathfrak{A}_+(I)$.
\end{enumerate}
The parity projections $P_\pm \coloneqq \frac{1}{2}(I \pm \alpha)$ yield the direct-sum $\mathbb{Z}_2$-grading:
\begin{equation}
\mathfrak{A}(I) = \mathfrak{A}_+(I) \oplus \mathfrak{A}_-(I).
\end{equation}
\end{definition}

\begin{definition}[Operator Word Decomposition]
A \emph{letter} is a pair $(x, \tau) \in \Lambda \times \{0, 1\}$, representing $\psi_x^\dagger$ (if $\tau = 1$) or $\psi_x$ (if $\tau = 0$). A \emph{word} of length $m \in \mathbb{N}$ is a finite sequence $w = [a_1, a_2, \dots, a_m]$ of letters, representing the operator monomial product:
\begin{equation}
O(w) \coloneqq O(a_1) O(a_2) \cdots O(a_m).
\end{equation}
A word $w$ is supported in region $I \subseteq \Lambda$ if all its letters have spatial coordinates in $I$.
\end{definition}

\begin{lemma}[Parity of Operator Words]
\label{lem:word_parity}
For any word $w$ of length $|w| = m$:
\begin{equation}
\alpha(O(w)) = (-1)^m O(w).
\end{equation}
Consequently, the homogeneous subspace $\mathfrak{A}_\sigma(I)$ is precisely the complex linear span of supported words $w$ whose length satisfies $|w| \equiv \sigma \pmod 2$.
\end{lemma}

\begin{proof}
Inserting $\Gamma^2=I$ between consecutive letters expresses $\alpha(O(w))$ as the product of the parity conjugates of its $m$ letters. Each conjugation contributes $-1$, hence
\begin{equation}
\alpha(O(w))=(-1)^m O(w).
\end{equation}
Since $\mathfrak{A}(I)$ is spanned by supported words, the parity projections $P_\pm$ show that $\mathfrak{A}_\sigma(I)$ is spanned by words with $|w|\equiv\sigma\pmod 2$.
\end{proof}

\section{The Haag--Kastler Axioms as Proved Theorems}

We now state and prove the constructive realizations of the Haag--Kastler axioms.

\begin{theorem}[Isotony and Additivity]
\label{thm:isotony_additivity}
For all regions $I, J \subseteq \Lambda$:
\begin{enumerate}
    \item \textbf{Isotony:} If $I \subseteq J$, then:
    \begin{equation}
    \mathfrak{A}(I) \subseteq \mathfrak{A}(J).
    \end{equation}
    \item \textbf{Additivity:} The algebra of the union is the supremum of the local algebras:
    \begin{equation}
    \mathfrak{A}(I \cup J) = \mathfrak{A}(I) \vee \mathfrak{A}(J),
    \end{equation}
    where $\vee$ denotes the generated subalgebra $\mathrm{AlgGen}(\mathfrak{A}(I) \cup \mathfrak{A}(J))$.
\end{enumerate}
\end{theorem}

\begin{proof}
If $I \subseteq J$, then the generating sets satisfy $\mathcal{G}(I) \subseteq \mathcal{G}(J)$. Because the operation of taking the generated subalgebra is monotonic, $\mathrm{AlgGen}(\mathcal{G}(I)) \subseteq \mathrm{AlgGen}(\mathcal{G}(J))$, establishing isotony.
For additivity, the generating set of $I \cup J$ is the union of generating sets: $\mathcal{G}(I \cup J) = \mathcal{G}(I) \cup \mathcal{G}(J)$. The subalgebra generated by $\mathcal{G}(I) \cup \mathcal{G}(J)$ is by definition the supremum $\mathfrak{A}(I) \vee \mathfrak{A}(J)$.
\end{proof}

\begin{theorem}[Twisted Locality / Graded Commutation]
\label{thm:twisted_locality}
Let $I, J \subseteq \Lambda$ be disjoint spatial regions ($I \cap J = \emptyset$). For any homogeneous elements $A \in \mathfrak{A}_\sigma(I)$ and $B \in \mathfrak{A}_\tau(J)$ with parities $\sigma, \tau \in \{0, 1\}$:
\begin{equation}
A B = (-1)^{\sigma \cdot \tau} B A.
\end{equation}
\end{theorem}

\begin{proof}
By Lemma~\ref{lem:word_parity}, it suffices by linearity to verify the identity when $A = O(w_1)$ and $B = O(w_2)$ are words supported in $I$ and $J$, respectively, with lengths $|w_1| = m \equiv \sigma \pmod 2$ and $|w_2| = n \equiv \tau \pmod 2$.

Let $w_1 = [a_1, \dots, a_m]$ and $w_2 = [b_1, \dots, b_n]$. For any letter $a_k$ (at site $x \in I$) and $b_\ell$ (at site $y \in J$), because $I \cap J = \emptyset$, we have $x \neq y$.
By Theorem~\ref{thm:spatial_car}, every letter supported in $I$ anticommutes with every letter supported in $J$. Commuting a word of length $m$ past a word of length $n$ therefore incurs exactly $mn$ minus signs, so
\begin{equation}
O(w_1)O(w_2)=(-1)^{mn}O(w_2)O(w_1)=(-1)^{\sigma\tau}O(w_2)O(w_1),
\end{equation}
where $m\equiv\sigma$ and $n\equiv\tau$ modulo $2$. Linearity completes the proof.
\end{proof}

\begin{corollary}[Strict Locality of Physical Observables]
\label{cor:observable_locality}
Let $I, J \subseteq \Lambda$ be disjoint regions. For any physical (even) observables $A \in \mathfrak{A}_+(I)$ and $B \in \mathfrak{A}_+(J)$:
\begin{equation}
[A, B] \coloneqq A B - B A = 0.
\end{equation}
\end{corollary}

\begin{proof}
Since $A$ and $B$ are even, $\sigma = 0$ and $\tau = 0$, so $\sigma \cdot \tau = 0$. Theorem~\ref{thm:twisted_locality} immediately yields $A B = (-1)^0 B A = B A$, so $[A, B] = 0$.
\end{proof}

\begin{remark}[Physical Context for Locality]
Corollary~\ref{cor:observable_locality} is the discrete lattice realization of Einstein causality: measurements performed in spatially disjoint regions commute and cannot interfere with one another. This graded commutation structure explains why only even operators qualify as physical observables. Fermionic fields $\psi_x$ are unobservable directly; only their even bilinears $\psi_x^\dagger \psi_x$, $\psi_x^\dagger \psi_y$ generate physically measurable gauge-invariant observables.
\end{remark}

\section{Global Algebra Irreducibility}

In quantum field theory, irreducibility ensures that the Fock representation contains no superselection sectors other than those governed by global symmetries, and that the field algebra generates all bounded operators.

\begin{theorem}[Global Irreducibility]
\label{thm:global_irreducibility}
Over the entire spatial lattice $\Lambda$, the local algebra generates the entire algebra of linear operators on Fock space:
\begin{equation}
\mathfrak{A}(\Lambda) = \mathrm{End}_{\mathbb{C}}(\mathcal{F}(L)).
\end{equation}
\end{theorem}

\begin{proof}
The operator space $\mathrm{End}_{\mathbb{C}}(\mathcal{F}(L))$ is spanned by the elementary rank-one matrix units $|S\rangle\langle T|$ for all configurations $S, T \subseteq \Lambda^*$. We prove that every such matrix unit belongs to $\mathfrak{A}(\Lambda)$:
\begin{enumerate}
    \item By Theorem~\ref{thm:op_fourier_inversion} (operator Fourier inversion), each momentum mode generator $c_k$ and $c_k^\dagger$ is a linear combination of spatial operators $\{\psi_x, \psi_x^\dagger \mid x \in \Lambda\}$. Therefore, $c_k \in \mathfrak{A}(\Lambda)$ and $c_k^\dagger \in \mathfrak{A}(\Lambda)$ for all $k \in \Lambda^*$.
    \item The orthogonal projection onto the empty vacuum state $|\emptyset\rangle$ is constructed as:
    \begin{equation}
    P_\emptyset \coloneqq \prod_{k \in \Lambda^*} (I - c_k^\dagger c_k) = |\emptyset\rangle\langle\emptyset|.
    \end{equation}
    Since each $c_k^\dagger c_k \in \mathfrak{A}(\Lambda)$, $P_\emptyset \in \mathfrak{A}(\Lambda)$.
    \item For configurations $S,T\subseteq\Lambda^*$, let $C(S)=\prod_{k\in S}^{\downarrow}c_k^\dagger$ be the descending creation string and $A(T)=\prod_{p\in T}^{\uparrow}c_p$ the ascending annihilation string. The ordered occupation basis fixes signs $\eta(S),\eta(T)\in\{+1,-1\}$ with $C(S)|\emptyset\rangle=\eta(S)|S\rangle$ and $\langle\emptyset|A(T)=\eta(T)\langle T|$, so $C(S)P_\emptyset A(T)=\eta(S)\eta(T)|S\rangle\langle T|$. Rescaling by this sign shows each matrix unit lies in $\mathfrak{A}(\Lambda)$.
\end{enumerate}
Because the matrix units span $\mathrm{End}_{\mathbb{C}}(\mathcal{F}(L))$, we conclude $\mathfrak{A}(\Lambda) = \mathrm{End}_{\mathbb{C}}(\mathcal{F}(L))$.
\end{proof}

\begin{remark}[Explicit Matrix Units]
The construction above establishes global irreducibility directly: ordered creation and annihilation strings together with the vacuum projector produce every rank-one matrix unit. This explicit algebraic argument does not rely on an abstract dimension-counting step.
\end{remark}

\section{Boundary Extensions: Holonomy, Twists, and Spatial Transport}
\label{sec:boundary_extensions}

In physical systems on a spatial ring, fermions may acquire a non-trivial boundary phase upon traversing the circle. This occurs in the presence of an Aharonov--Bohm magnetic flux through the ring, or when mapping 1D spin chains to fermions via Jordan--Wigner transformations.

The point of this extension is to distinguish local physics from the global
choice of how a wavefunction closes around the ring. A particle transported
once around the circle may return with phase $\tau$, even though the local
CAR relations at neighboring sites are unchanged. For a charged particle,
this phase records the flux through the ring; in a Jordan--Wigner map it
can encode the fermion-parity-dependent boundary condition. The corresponding
allowed momenta are shifted, so the twist changes momentum quantization and
the phase accumulated by a full circuit. It does not change the finite
one-particle space or create a new local degree of freedom. We track the
phase explicitly because later spatial translations and bosonic fields
must reproduce the same winding when their position coordinate is advanced
by one full circumference.

\subsection{Distinction Between Dual Space Modes and the Momentum Band}

A common source of confusion in the physics literature is the distinction between:
\begin{enumerate}
    \item \textbf{Modes in the abstract dual space:} The discrete Fourier modes remain indexed by the same discrete integers $k \in \{-h+1, \dots, h\}$. The single-particle state space dimension remains strictly $L$.
    \item \textbf{The physical momentum band:} In the presence of a boundary twist $\beta \in \mathbb{R}$, the physical crystal wavenumbers are shifted by the flux:
    \begin{equation}
    p_{\mathrm{phys}}(k) \coloneqq \frac{2\pi (k + \beta)}{L}.
    \end{equation}
    Under periodic boundary conditions ($\beta = 0$), $p_{\mathrm{phys}}(k) = \frac{2\pi k}{L}$. Under anti-periodic boundary conditions ($\beta = -1/2$), the wavenumbers are shifted by half a unit: $p_{\mathrm{phys}}(k) = \frac{2\pi (k - 1/2)}{L}$, rendering the band completely reflection-symmetric around zero.
\end{enumerate}

\subsection{The Boundary Condition Datum and Lifted Spatial Fields}

\begin{definition}[Boundary Datum $b$ and Boundary Twist]
A \emph{boundary condition datum} $b$ on the circular chain $\Lambda = \mathbb{Z}/L\mathbb{Z}$ consists of a pair:
\begin{equation}
b \coloneqq (r, \tau),
\end{equation}
where $r \in \mathbb{C}$ is a complex phase factor with $|r| = 1$ representing the phase acquired per spatial unit step, and $\tau \in U(1)$ is the total \emph{boundary holonomy} accumulated over a full circuit of length $L$:
\begin{equation}
\tau \coloneqq r^L.
\end{equation}
Parametrizing $r$ by a real flux / twist parameter $\beta \in \mathbb{R}$:
\begin{equation}
r = \exp\left(\frac{2\pi i \beta}{L}\right) \implies \tau = \exp(2\pi i \beta).
\end{equation}
Standard physical cases are:
\begin{itemize}
    \item \textbf{Periodic Boundary Conditions (PBC):} $\beta = 0 \implies r = 1, \tau = 1$.
    \item \textbf{Anti-Periodic Boundary Conditions (APBC):} $\beta = -\frac{1}{2} \implies r = e^{-\pi i / L}, \tau = -1$.
\end{itemize}
\end{definition}

\begin{definition}[Lifted Twisted Spatial Fields]
For an integer coordinate $n \in \mathbb{Z}$, let $n = w L + x$ be the unique Euclidean division with winding number $w \in \mathbb{Z}$ and residue $x \in \{0, \dots, L-1\}$. In formulas for Fourier characters we write $x$ in place of this residue, as established in Chapter~\ref{ch:fourier}.
The lifted twisted fermion field on $\mathbb{Z}$ associated with boundary datum $b = (r, \tau)$ is defined by:
\begin{equation}
\psi_b(n) \coloneqq r^n\, \psi_x.
\end{equation}
Shifting by a full circuit of length $L$ incurs the seam holonomy phase:
\begin{equation}
\psi_b(n + L) = r^{n+L} \psi_x = r^L (r^n \psi_x) = \tau\, \psi_b(n).
\end{equation}
\end{definition}

\begin{theorem}[Spatial Translation Unitary and Holonomy Action]
\label{thm:twisted_transport}
The translation theorem packages spatial covariance into one unitary on Fock space. On a one-particle momentum state, translating by $m$ sites multiplies the wave by its momentum phase; the boundary twist contributes the additional phase $r^{-m}$. Second quantization applies this phase to each occupied mode, so a full circuit records the boundary holonomy as a gauge rotation by total particle number.
For each spatial shift $m \in \mathbb{Z}$ and boundary datum $b = (r, \tau)$, there exists a unitary operator $T_b(m) \in \mathrm{End}_{\mathbb{C}}(\mathcal{F}(L))$ acting on momentum basis states $|S\rangle$ by:
\begin{equation}
T_b(m) |S\rangle \coloneqq r^{-m \#S} \zeta^{-m \sum_{k \in S} k} |S\rangle,
\end{equation}
such that spatial fields transform covariantly:
\begin{equation}
T_b(m) \psi_b(n) T_b(m)^\dagger = \psi_b(n + m).
\end{equation}
In particular, a full circuit translation $m = L$ evaluates to the total charge gauge transformation:
\begin{equation}
T_b(L) = \tau^{-N_{\mathrm{tot}}}.
\end{equation}
\end{theorem}

\begin{proof}
The momentum-basis action of $T_b(m)$ gives directly
\begin{equation}
T_b(m)c_kT_b(m)^\dagger=r^m\zeta^{mk}c_k.
\end{equation}
Using $\psi_b(n)=\frac{r^n}{\sqrt L}\sum_{k\in\Lambda^*}\zeta^{kn}c_k$, we obtain
\begin{equation}
T_b(m)\psi_b(n)T_b(m)^\dagger=\frac{r^n}{\sqrt L}\sum_k\zeta^{kn}r^m\zeta^{mk}c_k=\psi_b(n+m).
\end{equation}
For $m=L$, $\zeta^L=1$ and $r^L=\tau$, so $T_b(L)|S\rangle=\tau^{-\#S}|S\rangle$. Since $N_{\mathrm{tot}}\in\mathrm{End}_{\mathbb C}(\mathcal F(L))$ has eigenvalue $\#S$ on $|S\rangle$, this is precisely
\begin{equation}
T_b(L)=\tau^{-N_{\mathrm{tot}}}.
\end{equation}
\end{proof}

\begin{corollary}[Jordan--Wigner Sector Dictionary]
\label{cor:jw_dictionary}
Here the total particle number is the operator $N_{\mathrm{tot}}=\sum_{k\in\Lambda^*}c_k^\dagger c_k\in\mathrm{End}_{\mathbb C}(\mathcal F(L))$. When mapping a 1D spin-$1/2$ chain with periodic boundary conditions to lattice fermions via the Jordan--Wigner transformation, the effective boundary condition for the fermions depends dynamically on the total particle number:
\begin{equation}
\tau_{\mathrm{eff}} = - (-1)^{N_{\mathrm{tot}}}.
\end{equation}
Consequently:
\begin{itemize}
    \item Sectors with an \textbf{even} number of particles experience \textbf{antiperiodic} boundary conditions ($\tau = -1$).
    \item Sectors with an \textbf{odd} number of particles experience \textbf{periodic} boundary conditions ($\tau = +1$).
\end{itemize}
\end{corollary}

\begin{remark}[Physical Context for the Sector Dictionary]
Corollary~\ref{cor:jw_dictionary} explains why analyses of one-dimensional quantum field theories must accommodate general boundary holonomies. When solving the 1D quantum Ising or XY spin chain in a transverse field (Lieb, Schultz, and Mattis~\cite{lieb1961}), the Hamiltonian decomposes into two uncoupled parity blocks: the even-fermion parity block (which feels antiperiodic boundary conditions) and the odd-fermion parity block (which feels periodic boundary conditions). Accommodating arbitrary holonomies $\tau \in U(1)$ is therefore not a mathematical luxury, but a physical necessity for diagonalizing interacting quantum spin models.
\end{remark}

\section{Chapter Summary and Literature Comparison}

In this chapter, we formulated and proved the Haag--Kastler axioms of Algebraic Quantum Field Theory on the discrete lattice:
\begin{itemize}
    \item Provided precise definitions of operator algebras, $C^*$-subalgebras, and automorphisms for physics readers.
    \item Local field algebras $\mathfrak{A}(I)$ are generated by spatial fermions $\{\psi_x, \psi_x^\dagger \mid x \in I\}$.
    \item The global parity operator $\Gamma$ generates an involutive $*$-automorphism $\alpha$, yielding a direct-sum $\mathbb{Z}_2$-grading $\mathfrak{A}(I) = \mathfrak{A}_+(I) \oplus \mathfrak{A}_-(I)$.
    \item Proved the Haag--Kastler axioms: Isotony, Additivity, Twisted Locality ($A B = (-1)^{\sigma\cdot\tau} B A$), and Strict Locality of Physical Observables ($[A, B] = 0$ for $A, B \in \mathfrak{A}_+$).
    \item Proved Global Irreducibility $\mathfrak{A}(\Lambda) = \mathrm{End}_{\mathbb{C}}(\mathcal{F}(L))$ constructively via vacuum projectors and creation words.
    \item Formulated boundary extensions: explicitly distinguished dual space modes from shifted physical wavenumbers $p_{\mathrm{phys}} = \frac{2\pi(k+\beta)}{L}$, defined the boundary datum $b = (r, \tau)$, proved the covariance of spatial translation unitaries $T_b(m) \psi_b(n) T_b(m)^\dagger = \psi_b(n+m)$, and contextualized the Jordan--Wigner sector dictionary.
\end{itemize}

\begin{remark}[Comparison with Standard Continuum Literature]
In continuum AQFT (Haag~\cite{haag1996}), local nets $\mathfrak{A}(O)$ are assigned to open diamond regions in continuous Minkowski spacetime. Proving that an interacting continuum model satisfies the Haag--Kastler axioms requires profound functional analysis (e.g., Glimm--Jaffe constructive quantum field theory, Tomita--Takesaki modular theory, and Bisognano--Wichmann bisector theorems). 

On the discrete lattice, as demonstrated in this chapter, all Haag--Kastler axioms become constructive algebraic theorems on finite-dimensional matrices. By establishing strict locality and global irreducibility at the lattice level, we ensure that the bosonization dictionary developed in Chapters~\ref{ch:heisenberg_algebra} and~\ref{ch:haldane_completeness} is mathematically watertight before taking continuum approximations.
\end{remark}
