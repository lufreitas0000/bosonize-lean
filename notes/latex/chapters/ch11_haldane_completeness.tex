\chapter{Haldane Completeness}
\label{ch:haldane_completeness}

\section{Introduction and the Completeness Problem}

In Chapter~\ref{ch:density_modes} and Chapter~\ref{ch:heisenberg_algebra}, we established that the non-wrapping spatial density modes $\rho_m$ satisfy the exact $U(1)$ Kac--Moody (Heisenberg) current commutation relations:
\begin{equation}
[\rho_{-m}, \rho_n] \psi = m\, \delta_{m, n}\, \psi \quad \text{for all } \psi \in B(N, K),
\end{equation}
provided the system parameters satisfy the uniform margin condition $2M + K + |N| \le h$.
Algebraically, this implies that the operators $\rho_m$ ($m \ge 1$) act as independent bosonic creation operators, while $\rho_{-m}$ act as annihilation operators that annihilate the sector ground state $|N\rangle_0$.

However, an algebraic representation of an operator algebra on a Hilbert space does not automatically guarantee that the generated states exhaust the space. A subalgebra may act faithfully on an invariant subspace without spanning the whole carrier space. The central question of constructive bosonization is therefore:
\begin{quote}
\emph{Do the multi-mode bosonic density excitations generated by acting with $\{\rho_m\}_{m \ge 1}$ on the Fermi ground sea $|N\rangle_0$ completely span the low-energy fermionic Hilbert space?}
\end{quote}
In the continuum theory of one-dimensional Fermi liquids, this property is known as \emph{Haldane completeness}. In infinite-volume continuum models, completeness is typically argued via representation theory of the infinite-dimensional affine Lie algebra $\widehat{\mathfrak{u}}(1)$ or character formulas for the Virasoro algebra. On a finite discrete lattice, however, such continuum arguments break down: the global fermionic Fock space has finite dimension $2^{2h}$, which cannot be isomorphic to the infinite-dimensional Fock space of an infinite family of harmonic oscillators, and boundary-wrapping effects distort the current algebra at high energies.

In this chapter, we resolve Haldane completeness constructively and strictly within finite dimensions. We establish two complementary regimes:
\begin{enumerate}
    \item \textbf{Regime R1 (Combinatorial Rectangle Bijection):} At excitation energy $E$, whenever $E \le \min(h+N, h-N)$, the dimension of the fermionic fixed-energy subspace $\mathcal{H}^N_E$ is identically equal to the number of unrestricted integer partitions $p(E)$:
    \begin{equation}
    \dim_{\mathbb{C}} \mathcal{H}^N_E = p(E).
    \end{equation}
    \item \textbf{Regime R2 (Gram Orthogonality and Completeness):} Under the margin condition $2K + |N| \le h$, the bosonic partition states $|\lambda; N\rangle$ associated with partitions $\lambda \vdash E \le K$ are mutually orthogonal with strictly positive norms given by the Hall product $z_\lambda = \prod m^{r_m} r_m!$.
\end{enumerate}
Combining these two results proves that the bosonic density states form an explicit, complete orthogonal basis of the low-energy fermionic budget space $B(N, K)$.

\section{Fermionic Coordinates and the Rectangle Bijection (Regime R1)}

We begin by establishing the exact combinatorial dimension of the fermionic fixed-energy subspaces using the particle-hole displacement coordinates introduced in Chapter~\ref{ch:vacuum_budget}.

\begin{definition}[Fixed-Energy Subspace]
Let $h \ge 1$ and let $N \in \mathbb{Z}$ be an admissible sector charge ($-h \le N \le h$). For each excitation energy $E \ge 0$, the \emph{fixed-energy fermionic subspace} $\mathcal{H}^N_E$ is defined as the coordinate subspace spanned by occupation configurations of charge $N$ and excitation $e(S) = E$:
\begin{equation}
\mathcal{H}^N_E \coloneqq \operatorname{span}_{\mathbb{C}} \left\{ |S\rangle \mid S \subseteq B,\; \mathcal{N}(S) = N,\; e(S) = E \right\}.
\end{equation}
Because the occupation kets $\{|S\rangle\}$ form an orthonormal basis of the ambient Fock space $\mathcal{F}(L)$, the dimension $\dim_{\mathbb{C}} \mathcal{H}^N_E$ is precisely the number of subsets $S \subseteq B$ satisfying $\mathcal{N}(S) = N$ and $e(S) = E$.
\end{definition}

Recall from Chapter~\ref{ch:vacuum_budget} that an occupation configuration $S \subseteq B = \{-h+1, \dots, h\}$ with $|S| = n = h + N$ occupied modes is uniquely characterized by its sorted mode labels:
\begin{equation}
-h < s_0 < s_1 < \dots < s_{n-1} \le h.
\end{equation}
The reference ground configuration $\mathcal{S}_N$ of charge $N$ has consecutive occupied modes:
\begin{equation}
g_i = -h + 1 + i, \quad \text{for } 0 \le i < n.
\end{equation}
The displacement of the $i$-th particle is:
\begin{equation}
d_i \coloneqq s_i - g_i = s_i - (-h + 1 + i).
\end{equation}
As proved in Theorem~\ref{thm:rank_formula}, the sequence of displacements satisfies:
\begin{equation}
0 \le d_0 \le d_1 \le \dots \le d_{n-1} \le 2h - n,
\end{equation}
and the excitation energy is the sum of these displacements:
\begin{equation}
e(S) = \sum_{i=0}^{n-1} d_i.
\end{equation}

\begin{definition}[Bounded Partitions and Rectangle Embedding]
An integer partition $\lambda$ of an integer $E \ge 0$ is a multiset of positive integers whose sum is $E$.
Geometrically, a sequence of non-decreasing displacements $(d_0, d_1, \dots, d_{n-1})$ with $\sum d_i = E$ corresponds to a Young diagram of total area $E$ that fits inside a bounding rectangle of $n$ rows and $2h - n$ columns:
\begin{equation}
\text{rows} = n = h + N, \qquad \text{columns} = 2h - n = h - N.
\end{equation}
The non-zero entries of $(d_0, \dots, d_{n-1})$ constitute the parts of an integer partition $\lambda \vdash E$.
\end{definition}

\begin{lemma}[Explicit Configuration--Partition Bijection]
\label{lem:rectangle_bijection}
Let $h \ge 1$ and $-h \le N \le h$. The map $S \mapsto (d_0, \dots, d_{n-1})$ is a bijection between:
\begin{enumerate}
    \item Subsets $S \subseteq B$ with $|S| = h + N$ and $e(S) = E$, and
    \item Sequences of integers $(d_0, \dots, d_{n-1})$ satisfying $0 \le d_0 \le \dots \le d_{n-1} \le h - N$ and $\sum_{i=0}^{n-1} d_i = E$.
\end{enumerate}
\end{lemma}

\begin{proof}
Given $S \subseteq B$ with $|S| = n$, the sorted elements $s_0 < \dots < s_{n-1}$ are uniquely determined. Since $s_{i+1} > s_i$, we have $s_{i+1} - s_i \ge 1$, which gives:
\[
d_{i+1} - d_i = (s_{i+1} - g_{i+1}) - (s_i - g_i) = (s_{i+1} - s_i) - 1 \ge 0.
\]
Hence $d_0 \le d_1 \le \dots \le d_{n-1}$. The lower bound $d_0 \ge 0$ follows from $s_0 \ge -h + 1 = g_0$. The upper bound follows from $s_{n-1} \le h$, which yields:
\[
d_{n-1} = s_{n-1} - g_{n-1} = s_{n-1} - (-h + n) \le h - (-h + n) = 2h - n = h - N.
\]
Conversely, given any sequence $(d_0, \dots, d_{n-1})$ satisfying $0 \le d_0 \le \dots \le d_{n-1} \le h - N$, we define:
\[
s_i \coloneqq -h + 1 + i + d_i \quad (0 \le i < n).
\]
Then $s_{i+1} - s_i = 1 + (d_{i+1} - d_i) \ge 1$, so the values $s_i$ are strictly increasing integers. Furthermore:
\[
s_0 = -h + 1 + d_0 \ge -h + 1, \qquad s_{n-1} = -h + n + d_{n-1} \le -h + n + (2h - n) = h.
\]
Thus, every $s_i \in B$, and setting $S = \{s_0, \dots, s_{n-1}\}$ yields a subset of cardinality $n = h + N$ with excitation $\sum d_i = E$. These two constructions are mutually inverse.
\end{proof}

\begin{theorem}[Regime R1: Exact Dimension Equality]
\label{thm:regime_r1}
Let $h \ge 1$, $-h \le N \le h$, and let $E \ge 0$. Assume the \emph{Regime R1 margin condition}:
\begin{equation}
E \le \min(h + N, \; h - N) = h - |N|.
\end{equation}
Then the dimension of the fixed-energy subspace $\mathcal{H}^N_E$ is identically equal to the unrestricted partition function $p(E)$:
\begin{equation}
\dim_{\mathbb{C}} \mathcal{H}^N_E = p(E).
\end{equation}
\end{theorem}

\begin{proof}
By Lemma~\ref{lem:rectangle_bijection}, $\dim_{\mathbb{C}} \mathcal{H}^N_E$ equals the number of partitions of $E$ whose Young diagram fits inside the $(h+N) \times (h-N)$ rectangle.
Let $\lambda = (\lambda_1 \ge \lambda_2 \ge \dots \ge \lambda_\ell > 0)$ be an arbitrary unrestricted integer partition of $E$, so that $\sum_{j=1}^\ell \lambda_j = E$ and each $\lambda_j \ge 1$.
\begin{enumerate}
    \item The number of parts $\ell$ satisfies $\ell \le \sum_{j=1}^\ell \lambda_j = E$. Under the margin condition, $E \le h + N = n$, so $\ell \le n$. Thus, padding $\lambda$ with $n - \ell \ge 0$ zeros yields a non-decreasing sequence $(d_0, \dots, d_{n-1})$ of length $n$.
    \item The largest part $\lambda_1$ satisfies $\lambda_1 \le \sum_{j=1}^\ell \lambda_j = E$. Under the margin condition, $E \le h - N$, so $\lambda_1 \le h - N$. Thus, every part fits within the column limit $2h - n = h - N$.
\end{enumerate}
Because $E \le \min(h+N, h-N)$, \emph{every} unrestricted partition of $E$ automatically fits inside the bounding box. None of the boundary constraints are violated.
Therefore, the restriction of bounded partitions is completely non-binding, and the set of valid fermionic configurations at energy $E$ is in bijective correspondence with the unrestricted partitions of $E$. We conclude:
\[
\dim_{\mathbb{C}} \mathcal{H}^N_E = p(E).
\]
\end{proof}

\section{Bosonic Partition States}

Having determined the exact dimension of the fermionic target spaces $\mathcal{H}^N_E$, we now construct the candidate bosonic basis states.

\begin{definition}[Bosonic Partition States]
Let $E \ge 0$ and let $\lambda \vdash E$ be an integer partition of $E$. Represent $\lambda$ as a multiset of positive integer parts $\{m_1, m_2, \dots, m_r\}$ sorted in non-increasing order:
\begin{equation}
m_1 \ge m_2 \ge \dots \ge m_r \ge 1, \qquad \sum_{j=1}^r m_j = E.
\end{equation}
Alternatively, let $r_m \ge 0$ denote the multiplicity of part $m \ge 1$ in $\lambda$.
We define the \emph{bosonic partition state} $|\lambda; N\rangle \in \mathcal{F}(L)$ by the ordered action of the positive density modes on the sector ground state:
\begin{equation}
|\lambda; N\rangle \coloneqq \left( \prod_{j=1}^r \rho_{m_j} \right) |N\rangle_0 = \left( \prod_{m \ge 1} \rho_m^{r_m} \right) |N\rangle_0.
\end{equation}
For $E = 0$, the unique partition is the empty partition $\emptyset \vdash 0$, and $|\emptyset; N\rangle \coloneqq |N\rangle_0$.
\end{definition}

\begin{lemma}[Order Independence and Energy Grading]
\label{lem:partition_state_well_defined}
The partition state $|\lambda; N\rangle$ satisfies the following properties:
\begin{enumerate}
    \item \textbf{Order Independence:} For any permutation $\sigma \in S_r$ of the parts,
    \begin{equation}
    \rho_{m_{\sigma(1)}} \rho_{m_{\sigma(2)}} \cdots \rho_{m_{\sigma(r)}} |N\rangle_0 = |\lambda; N\rangle.
    \end{equation}
    \item \textbf{Exact Energy Grading:}
    \begin{equation}
    |\lambda; N\rangle \in \mathcal{H}^N_E.
    \end{equation}
\end{enumerate}
\end{lemma}

\begin{proof}
\begin{enumerate}
    \item By Theorem~\ref{thm:same_sign_comm}, positive density modes globally commute on the entire Fock carrier:
    \[
    [\rho_m, \rho_n] = 0 \quad \text{for all } m, n \ge 1.
    \]
    Any permutation of the operators in a written product of positive modes can be generated by adjacent transpositions $\rho_m \rho_n = \rho_n \rho_m$. Hence the product operator is completely independent of the ordering of the parts.
    \item By Theorem~\ref{thm:charge_energy_covariance}, each mode $\rho_m$ shifts the relative charge by $0$ and shifts the excitation energy by $+m$:
    \[
    \hat{N} \rho_m = \rho_m \hat{N}, \qquad [\hat{P}, \rho_m] = m \rho_m.
    \]
    Since $|N\rangle_0 \in \mathcal{H}^N_0$, acting with $\rho_{m_r}$ produces a state in $\mathcal{H}^N_{m_r}$. By induction on $r$, the composite state $|\lambda; N\rangle$ has charge $N$ and exact excitation energy:
    \[
    \sum_{j=1}^r m_j = E.
    \]
    Therefore, $|\lambda; N\rangle \in \mathcal{H}^N_E$.
\end{enumerate}
\end{proof}

\section{Regime R2: The Hall--Gram Matrix and Bosonic Orthogonality}

To prove that the partition states $|\lambda; N\rangle$ form a basis of $\mathcal{H}^N_E$, it remains to show that they are linearly independent. We do this by computing their inner products explicitly and showing that their Gram matrix is strictly positive-definite.

\begin{definition}[Hall Inner Product Factor]
For an integer partition $\lambda$ with part multiplicities $(r_m)_{m \ge 1}$, the \emph{Hall normalization factor} $z_\lambda$ is defined by:
\begin{equation}
z_\lambda \coloneqq \prod_{m \ge 1} m^{r_m}\, r_m!.
\end{equation}
Notice that since $m \ge 1$ and $r_m \ge 0$, every factor $m^{r_m} r_m! \ge 1$, so $z_\lambda \ge 1$ is a strictly positive integer.
\end{definition}

To evaluate $\langle \lambda; N \mid \mu; N \rangle$, we use the Hilbert adjoint property $\rho_m^\dagger = \rho_{-m}$ established in Lemma~\ref{lem:density_vanishing_adj}:
\begin{equation}
\langle \lambda; N \mid \mu; N \rangle = \left\langle N \right|_0 \left( \prod_{j=1}^r \rho_{m_j} \right)^\dagger \left( \prod_{k=1}^s \rho_{n_k} \right) |N\rangle_0 = \left\langle N \right|_0 \left( \prod_{j=1}^r \rho_{-m_j} \right) \left( \prod_{k=1}^s \rho_{n_k} \right) |N\rangle_0.
\end{equation}
This inner product can be computed by commuting lowering currents $\rho_{-m}$ successively through the word of raising currents to the right until they annihilate the ground state $|N\rangle_0$.

\begin{lemma}[Right-Remainder Margin Invariant]
\label{lem:right_remainder_margin}
Let $K \ge 1$ and assume the \emph{Regime R2 margin condition}:
\begin{equation}
2K + |N| \le h.
\end{equation}
Let $w = (n_1, n_2, \dots, n_s)$ be a sequence of positive integers with $\sum_{k=1}^s n_k \le K$, and let $m \le K$.
When $\rho_{-m}$ commutes past $\rho_{n_1}$, the remaining word $w' = (n_2, \dots, n_s)$ has energy:
\begin{equation}
E_{\mathrm{rem}} = \sum_{k=2}^s n_k \le K - n_1.
\end{equation}
Then the joint excursion for the commutator $[\rho_{-m}, \rho_{n_1}]$ satisfies:
\begin{equation}
m + n_1 + E_{\mathrm{rem}} + |N| \le m + n_1 + (K - n_1) + |N| = m + K + |N| \le 2K + |N| \le h.
\end{equation}
Consequently, the input state $W(w') |N\rangle_0$ satisfies the M2 (and M1) margin condition of Theorem~\ref{thm:kac_moody}.
\end{lemma}

\begin{proof}
The bound $m + n_1 + E_{\mathrm{rem}} + |N| \le m + K + |N|$ holds identically because $n_1 + E_{\mathrm{rem}} = \sum_{k=1}^s n_k \le K$.
Since $m \le K$, we have $m + K + |N| \le 2K + |N| \le h$.
Therefore, the remainder state $W(w') |N\rangle_0 \in B(N, E_{\mathrm{rem}})$ lies safely within the buffer zone where the Kac--Moody commutator is exact:
\[
[\rho_{-m}, \rho_{n_1}] W(w') |N\rangle_0 = m\, \delta_{m, n_1}\, W(w') |N\rangle_0.
\]
\end{proof}

\begin{lemma}[Lowering Contraction Formula]
\label{lem:lowering_contraction}
Let $K \ge 1$, $2K + |N| \le h$, and let $1 \le m \le K$.
For any partition state word $W(w) |N\rangle_0$ with positive parts and total energy $E \le K$:
\begin{equation}
\rho_{-m} W(w) |N\rangle_0 = m \cdot r_m(w) \cdot W(w \setminus \{m\}) |N\rangle_0,
\end{equation}
where $r_m(w)$ is the multiplicity of $m$ in $w$, and $w \setminus \{m\}$ is the word with one instance of $m$ deleted (with the right-hand side evaluating to zero if $m \notin w$).
\end{lemma}

\begin{proof}
We proceed by induction on the length $s$ of the word $w = (n_1, \dots, n_s)$.
\begin{enumerate}
    \item \textbf{Base step ($s = 0$):} $w = \emptyset$, so $W(\emptyset)|N\rangle_0 = |N\rangle_0$.
    By Theorem~\ref{thm:budget_action_annihilation}(2), $\rho_{-m} |N\rangle_0 = 0$. Since $r_m(\emptyset) = 0$, both sides equal zero.
    \item \textbf{Inductive step:} Let $w = (n_1, w')$. Then $W(w) = \rho_{n_1} W(w')$.
    By Lemma~\ref{lem:right_remainder_margin}, we may commute $\rho_{-m}$ past $\rho_{n_1}$:
    \[
    \rho_{-m} \rho_{n_1} W(w') |N\rangle_0 = \rho_{n_1} \rho_{-m} W(w') |N\rangle_0 + [\rho_{-m}, \rho_{n_1}] W(w') |N\rangle_0.
    \]
    By Lemma~\ref{lem:right_remainder_margin}, the commutator evaluates to:
    \[
    [\rho_{-m}, \rho_{n_1}] W(w') |N\rangle_0 = m\, \delta_{m, n_1}\, W(w') |N\rangle_0.
    \]
    Applying the induction hypothesis to $\rho_{-m} W(w') |N\rangle_0$ yields:
    \[
    \rho_{n_1} \left( m\, r_m(w')\, W(w' \setminus \{m\}) |N\rangle_0 \right) + m\, \delta_{m, n_1}\, W(w') |N\rangle_0.
    \]
    Since positive modes commute, $\rho_{n_1} W(w' \setminus \{m\}) = W((n_1, w' \setminus \{m\})) = W(w \setminus \{m\})$.
    Factoring out $m W(w \setminus \{m\}) |N\rangle_0$ gives the total coefficient:
    \[
    m \left( r_m(w') + \delta_{m, n_1} \right) = m \cdot r_m(w).
    \]
    This completes the induction.
\end{enumerate}
\end{proof}

\begin{theorem}[Regime R2: Hall--Gram Matrix Diagonalization]
\label{thm:hall_gram}
Let $h \ge 1$, $-h \le N \le h$, and $K \ge 0$. Under the Regime R2 margin condition:
\begin{equation}
2K + |N| \le h,
\end{equation}
the bosonic partition states satisfy the exact orthogonality relation for all $E, E' \le K$ and all partitions $\lambda \vdash E$, $\mu \vdash E'$:
\begin{equation}
\langle \lambda; N \mid \mu; N \rangle = \delta_{\lambda, \mu}\, z_\lambda = \delta_{\lambda, \mu} \prod_{m \ge 1} m^{r_m}\, r_m!.
\end{equation}
\end{theorem}

\begin{proof}
If $E \neq E'$, the states $|\lambda; N\rangle \in \mathcal{H}^N_E$ and $|\mu; N\rangle \in \mathcal{H}^N_{E'}$ belong to distinct eigenspaces of the excitation energy observable $\hat{E}$. By Definition~\ref{def:budget_subspace}, distinct energy subspaces $\mathcal{H}(N, E)$ and $\mathcal{H}(N, E')$ are spanned by disjoint subsets of the orthonormal occupation basis $\{|S\rangle\}$, hence are mutually orthogonal, so $\langle \lambda; N \mid \mu; N \rangle = 0$.

Now assume $E = E'$ and let $\lambda, \mu \vdash E$. We write $|\lambda; N\rangle = \rho_{m_1} \dots \rho_{m_r} |N\rangle_0$ and compute:
\[
\langle \lambda; N \mid \mu; N \rangle = \left\langle \lambda \setminus \{m_1\}; N \right| \rho_{-m_1} |\mu; N\rangle.
\]
By Lemma~\ref{lem:lowering_contraction},
\[
\rho_{-m_1} |\mu; N\rangle = m_1 \cdot r_{m_1}(\mu) \cdot |\mu \setminus \{m_1\}; N\rangle.
\]
If $m_1 \notin \mu$, then $r_{m_1}(\mu) = 0$, which immediately gives $\langle \lambda; N \mid \mu; N \rangle = 0$.
If $m_1 \in \mu$, we obtain:
\[
\langle \lambda; N \mid \mu; N \rangle = m_1 \cdot r_{m_1}(\mu) \cdot \langle \lambda \setminus \{m_1\}; N \mid \mu \setminus \{m_1\}; N \rangle.
\]
Proceeding by induction on the total number of parts, if $\lambda \neq \mu$, at some step a part of $\lambda$ fails to appear in the remaining multiset of $\mu$, yielding zero.
If $\lambda = \mu$, each occurrence of part $m$ is extracted successively, contributing factors $m \cdot r_m, m \cdot (r_m - 1), \dots, m \cdot 1 = m^{r_m} r_m!$.
The base case is $\langle N \mid N \rangle_0 = 1$.
Taking the product over all distinct part values $m \ge 1$ yields:
\[
\langle \lambda; N \mid \lambda; N \rangle = \prod_{m \ge 1} m^{r_m} r_m! = z_\lambda.
\]
\end{proof}

\begin{corollary}[Linear Independence]
\label{cor:partition_linear_independence}
Under the Regime R2 condition $2K + |N| \le h$, the family of partition states:
\begin{equation}
\left\{ |\lambda; N\rangle \mid \lambda \vdash E,\; 0 \le E \le K \right\}
\end{equation}
is mutually orthogonal and linearly independent over $\mathbb{C}$.
\end{corollary}

\begin{proof}
Suppose $\sum_{\lambda} c_\lambda |\lambda; N\rangle = 0$ for some complex coefficients $c_\lambda$.
Taking the inner product with $|\mu; N\rangle$ and using Theorem~\ref{thm:hall_gram} yields:
\[
0 = \left\langle \mu; N \;\middle|\; \sum_\lambda c_\lambda |\lambda; N\rangle \right\rangle = \sum_\lambda c_\lambda \delta_{\lambda, \mu} z_\lambda = c_\mu z_\mu.
\]
Since $z_\mu \ge 1 > 0$, we conclude $c_\mu = 0$ for all $\mu$.
\end{proof}

\section{Haldane Completeness and the Whole Budget Basis}

We now combine the counting from Regime R1 with the orthogonality from Regime R2 to prove the main theorems of this chapter.

\begin{lemma}[R2 Implies R1]
\label{lem:r2_implies_r1}
For any $K \ge 0$ and $-h \le N \le h$, if the Regime R2 margin condition holds:
\begin{equation}
2K + |N| \le h,
\end{equation}
then the Regime R1 margin condition holds for all $E \le K$:
\begin{equation}
E \le h - |N|.
\end{equation}
\end{lemma}

\begin{proof}
Since $E \le K$ and $K \ge 0$, we have $E \le 2K$. Therefore:
\[
E + |N| \le 2K + |N| \le h \implies E \le h - |N| = \min(h + N, h - N).
\]
\end{proof}

\begin{theorem}[Haldane Completeness at Fixed Energy]
\label{thm:haldane_completeness_fixed_energy}
Let $h \ge 1$, $-h \le N \le h$, and let $E \ge 0$ satisfy the margin condition:
\begin{equation}
2E + |N| \le h.
\end{equation}
Then the bosonic partition states $\{|\lambda; N\rangle \mid \lambda \vdash E\}$ form a complete orthogonal basis of the fermionic fixed-energy subspace $\mathcal{H}^N_E$:
\begin{equation}
\mathcal{H}^N_E = \operatorname{span}_{\mathbb{C}} \left\{ |\lambda; N\rangle \mid \lambda \vdash E \right\}, \qquad \dim_{\mathbb{C}} \mathcal{H}^N_E = p(E).
\end{equation}
\end{theorem}

\begin{proof}
We check the three requirements of a basis:
\begin{enumerate}
    \item \textbf{Subspace membership:} By Lemma~\ref{lem:partition_state_well_defined}(2), $|\lambda; N\rangle \in \mathcal{H}^N_E$ for every partition $\lambda \vdash E$.
    \item \textbf{Linear independence:} By Corollary~\ref{cor:partition_linear_independence}, the family $\{|\lambda; N\rangle \mid \lambda \vdash E\}$ is orthogonal with strictly positive norms, hence linearly independent over $\mathbb{C}$.
    \item \textbf{Dimension match:} The number of vectors in this linearly independent family is the number of partitions of $E$, which is $p(E)$.
    By Lemma~\ref{lem:r2_implies_r1}, the condition $2E + |N| \le h$ implies the Regime R1 condition $E \le h - |N|$.
    By Theorem~\ref{thm:regime_r1}, $\dim_{\mathbb{C}} \mathcal{H}^N_E = p(E)$.
\end{enumerate}
A linearly independent family of $p(E)$ vectors in a vector space of dimension $p(E)$ spans the entire space.
Therefore, the partition states span $\mathcal{H}^N_E$ and constitute an orthogonal basis.
\end{proof}

\begin{theorem}[The Whole Budget Basis]
\label{thm:whole_budget_basis}
Let $h \ge 1$, $-h \le N \le h$, and $K \ge 0$. Under the Regime R2 margin condition:
\begin{equation}
2K + |N| \le h,
\end{equation}
the disjoint union of partition states up to energy $K$:
\begin{equation}
\mathcal{B}_{\mathrm{basis}}(N, K) \coloneqq \bigcup_{E=0}^K \left\{ |\lambda; N\rangle \mid \lambda \vdash E \right\}
\end{equation}
forms a complete orthogonal basis of the full low-energy budget subspace $B(N, K)$.
In particular:
\begin{equation}
B(N, K) = \bigoplus_{E=0}^K \mathcal{H}^N_E, \qquad \dim_{\mathbb{C}} B(N, K) = \sum_{E=0}^K p(E).
\end{equation}
\end{theorem}

\begin{proof}
Recall from Definition~\ref{def:budget_subspace} that the budget subspace is:
\[
B(N, K) = \operatorname{span}_{\mathbb{C}} \left\{ |S\rangle \mid \mathcal{N}(S) = N,\; e(S) \le K \right\}.
\]
Since excitation energy is an integer taking values $0 \le E \le K$, the coordinate basis decomposes as:
\[
B(N, K) = \bigoplus_{E=0}^K \mathcal{H}^N_E.
\]
For each $E \le K$, the condition $2K + |N| \le h$ implies $2E + |N| \le 2K + |N| \le h$.
By Theorem~\ref{thm:haldane_completeness_fixed_energy}, the family $\{|\lambda; N\rangle \mid \lambda \vdash E\}$ is an orthogonal basis of $\mathcal{H}^N_E$.
Since spaces $\mathcal{H}^N_E$ and $\mathcal{H}^N_{E'}$ with $E \neq E'$ are mutually orthogonal, the union $\mathcal{B}_{\mathrm{basis}}(N, K)$ is an orthogonal basis of the direct sum $\bigoplus_{E=0}^K \mathcal{H}^N_E = B(N, K)$.
Its dimension is the sum of dimensions $\sum_{E=0}^K p(E)$.
\end{proof}
